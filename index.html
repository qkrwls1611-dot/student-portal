/** * StudentReason.gs * * 학생 사유서 신청 기능 * * 사유서신청 시트 헤더: * 신청일시 | 강의실 | 신청ID | 학생ID | 학생명 | 좌석번호 | * 사유일 | 사유유형 | 교시 | 사유내용 | 증빙서류 | 증빙제출일시 | * 처리상태 | 처리일시 | 처리자 | 관리자메모 */ /** * 학생 사유서 제출 */ function submitStudentReason( studentId, reasonDate, reasonType, startPeriod, endPeriod, reasonContent ) { const id = String(studentId || "") .trim(); const dateText = String(reasonDate || "") .trim(); const type = String(reasonType || "") .trim(); const startPeriodText = String(startPeriod || "") .trim(); const endPeriodText = String(endPeriod || "") .trim(); const content = String(reasonContent || "") .trim(); if (!id) { return { success: false, message: "학생ID가 없습니다." }; } if (!dateText) { return { success: false, message: "사유일을 선택해주세요." }; } const allowedTypes = [ "지각", "조퇴", "외출", "결석" ]; if ( allowedTypes.indexOf(type) === -1 ) { return { success: false, message: "사유유형을 선택해주세요." }; } if ( (type === "지각" || type === "외출") && ( !startPeriodText || !endPeriodText ) ) { return { success: false, message: "시작 교시와 종료 교시를 선택해주세요." }; } if ( type === "조퇴" && !startPeriodText ) { return { success: false, message: "조퇴 교시를 선택해주세요." }; } if (!content) { return { success: false, message: "사유내용을 입력해주세요." }; } if (content.length > 1000) { return { success: false, message: "사유내용은 1,000자 이내로 작성해주세요." }; } const studentResult = getStudentDashboardData( id ); if ( !studentResult || !studentResult.success ) { return { success: false, message: "학생 정보를 찾을 수 없습니다." }; } const student = studentResult.student; const ss = SpreadsheetApp .getActiveSpreadsheet(); const sheet = ss.getSheetByName( "사유서신청" ); if (!sheet) { throw new Error( "사유서신청 시트를 찾을 수 없습니다." ); } validateReasonHeaders_( sheet ); const now = new Date(); const applicationId = createReasonApplicationId_( id, now ); const rowData = { "신청일시": now, "강의실": student.classroom || "", "신청ID": applicationId, "학생ID": student.studentId || id, "학생명": student.name || "", "좌석번호": student.seat || "", "사유일": dateText, "사유유형": type, "교시": type === "결석" ? "" : ( type === "조퇴" ? startPeriodText : startPeriodText + " ~ " + endPeriodText ), "사유내용": content, "증빙서류": "", "증빙제출일시": "", "처리상태": "접수", "처리일시": "", "처리자": "", "관리자메모": "" }; appendReasonRowByHeaders_( sheet, rowData ); return { success: true, applicationId: applicationId, message: "사유서 제출이 완료되었습니다.", notice: "사유서 제출이 완료되었습니다.\n" + "증빙서류는 당일 제출하지 않으셔도 됩니다.\n" + "준비되는 대로 학생포털에서 증빙서류를 추가 제출해주세요." }; } /** * 학생 본인의 사유서 신청 내역 조회 * 최근 20건 */ function getStudentReasonList( studentId ) { const id = String(studentId || "") .trim(); if (!id) { return { success: false, message: "학생ID가 없습니다." }; } const ss = SpreadsheetApp .getActiveSpreadsheet(); const sheet = ss.getSheetByName( "사유서신청" ); if (!sheet) { throw new Error( "사유서신청 시트를 찾을 수 없습니다." ); } const lastRow = sheet.getLastRow(); const lastColumn = sheet.getLastColumn(); if (lastRow < 2) { return { success: true, rows: [] }; } const values = sheet .getRange( 1, 1, lastRow, lastColumn ) .getDisplayValues(); const headers = getReasonHeaderMap_( values[0] ); [ "신청일시", "신청ID", "학생ID", "사유일", "사유유형", "교시", "사유내용", "증빙서류", "증빙제출일시", "처리상태", "관리자메모" ].forEach( function(header) { if ( headers[header] === undefined ) { throw new Error( "사유서신청 시트에 '" + header + "' 헤더가 없습니다." ); } } ); const rows = []; for ( let i = 1; i < values.length; i++ ) { const rowId = String( values[i][ headers["학생ID"] ] || "" ).trim(); if (rowId !== id) { continue; } rows.push({ applicationId: String( values[i][ headers["신청ID"] ] || "" ).trim(), submittedAt: String( values[i][ headers["신청일시"] ] || "" ).trim(), reasonDate: String( values[i][ headers["사유일"] ] || "" ).trim(), reasonType: String( values[i][ headers["사유유형"] ] || "" ).trim(), period: String( values[i][ headers["교시"] ] || "" ).trim(), reasonContent: String( values[i][ headers["사유내용"] ] || "" ).trim(), evidenceUrl: String( values[i][ headers["증빙서류"] ] || "" ).trim(), evidenceSubmittedAt: String( values[i][ headers["증빙제출일시"] ] || "" ).trim(), status: String( values[i][ headers["처리상태"] ] || "" ).trim(), adminMemo: String( values[i][ headers["관리자메모"] ] || "" ).trim() }); } rows.reverse(); return { success: true, rows: rows.slice( 0, 20 ) }; } /** * 증빙서류 업로드 완료 후 기존 신청건에 링크 연결 * * 실제 파일 업로드 화면은 이후 별도 Apps Script HTML로 연결 */ function attachReasonEvidence_( studentId, applicationId, evidenceUrl ) { const id = String(studentId || "") .trim(); const requestId = String(applicationId || "") .trim(); const url = String(evidenceUrl || "") .trim(); if ( !id || !requestId || !url ) { return { success: false, message: "증빙서류 연결 정보가 올바르지 않습니다." }; } const ss = SpreadsheetApp .getActiveSpreadsheet(); const sheet = ss.getSheetByName( "사유서신청" ); if (!sheet) { throw new Error( "사유서신청 시트를 찾을 수 없습니다." ); } const values = sheet .getDataRange() .getDisplayValues(); const headers = getReasonHeaderMap_( values[0] ); const studentIdColumn = headers["학생ID"]; const applicationIdColumn = headers["신청ID"]; const evidenceColumn = headers["증빙서류"]; const evidenceAtColumn = headers["증빙제출일시"]; if ( studentIdColumn === undefined || applicationIdColumn === undefined || evidenceColumn === undefined || evidenceAtColumn === undefined ) { throw new Error( "사유서신청 시트의 증빙 관련 헤더를 확인해주세요." ); } for ( let i = 1; i < values.length; i++ ) { const rowStudentId = String( values[i][ studentIdColumn ] || "" ).trim(); const rowRequestId = String( values[i][ applicationIdColumn ] || "" ).trim(); if ( rowStudentId === id && rowRequestId === requestId ) { sheet .getRange( i + 1, evidenceColumn + 1 ) .setValue( url ); sheet .getRange( i + 1, evidenceAtColumn + 1 ) .setValue( new Date() ); return { success: true, message: "증빙서류가 등록되었습니다." }; } } return { success: false, message: "사유서 신청내역을 찾을 수 없습니다." }; } /** * 신청ID 생성 */ function createReasonApplicationId_( studentId, date ) { const timeText = Utilities .formatDate( date, "Asia/Seoul", "yyyyMMddHHmmss" ); const random = Utilities .getUuid() .replace(/-/g, "") .substring( 0, 6 ) .toUpperCase(); const idTail = String(studentId || "") .replace( /[^0-9A-Za-z]/g, "" ) .slice(-6); return ( "RSN-" + timeText + "-" + idTail + "-" + random ); } /** * 헤더 검증 */ function validateReasonHeaders_( sheet ) { const headers = sheet .getRange( 1, 1, 1, sheet.getLastColumn() ) .getDisplayValues()[0]; const map = getReasonHeaderMap_( headers ); const required = [ "신청일시", "강의실", "신청ID", "학생ID", "학생명", "좌석번호", "사유일", "사유유형", "교시", "사유내용", "증빙서류", "증빙제출일시", "처리상태", "처리일시", "처리자", "관리자메모" ]; required.forEach( function(header) { if ( map[header] === undefined ) { throw new Error( "사유서신청 시트에 '" + header + "' 헤더가 없습니다." ); } } ); } /** * 헤더 순서대로 행 추가 */ function appendReasonRowByHeaders_( sheet, rowData ) { const headers = sheet .getRange( 1, 1, 1, sheet.getLastColumn() ) .getDisplayValues()[0]; const row = headers.map( function(header) { const key = String(header || "") .trim(); return ( rowData[key] !== undefined ? rowData[key] : "" ); } ); sheet.appendRow( row ); } function getReasonHeaderMap_( headerRow ) { const map = {}; headerRow.forEach( function(value, index) { const header = String(value || "") .trim(); if (header) { map[header] = index; } } ); return map; } /** * ============================== * 사유서 증빙서류 업로드 * ============================== * * 최초 1회: * 1. Google Drive에 "사유서 증빙자료" 폴더를 생성 * 2. 아래 REASON_EVIDENCE_FOLDER_ID에 폴더 ID 입력 * * 허용: JPG / JPEG / PNG / PDF * 최대: 10MB */ const REASON_EVIDENCE_FOLDER_ID = "여기에_구글드라이브_폴더ID_입력"; function uploadStudentReasonEvidence( studentId, applicationId, fileName, mimeType, base64Data ) { const id = String(studentId || "").trim(); const requestId = String(applicationId || "").trim(); const name = String(fileName || "").trim(); const type = String(mimeType || "").trim(); const data = String(base64Data || "").trim(); if ( !id || !requestId || !name || !type || !data ) { return { success: false, message: "증빙서류 정보가 올바르지 않습니다." }; } const allowedTypes = [ "image/jpeg", "image/png", "application/pdf" ]; if ( allowedTypes.indexOf(type) === -1 ) { return { success: false, message: "JPG, PNG, PDF 파일만 제출할 수 있습니다." }; } const bytes = Utilities.base64Decode(data); if ( bytes.length > 10 * 1024 * 1024 ) { return { success: false, message: "증빙서류는 10MB 이하만 제출할 수 있습니다." }; } // 본인 신청건인지 먼저 확인 const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName( "사유서신청" ); if (!sheet) { throw new Error( "사유서신청 시트를 찾을 수 없습니다." ); } const values = sheet .getDataRange() .getDisplayValues(); const headers = getReasonHeaderMap_( values[0] ); let found = false; for ( let i = 1; i < values.length; i++ ) { const rowStudentId = String( values[i][ headers["학생ID"] ] || "" ).trim(); const rowApplicationId = String( values[i][ headers["신청ID"] ] || "" ).trim(); if ( rowStudentId === id && rowApplicationId === requestId ) { found = true; break; } } if (!found) { return { success: false, message: "본인의 사유서 신청내역을 찾을 수 없습니다." }; } if ( !REASON_EVIDENCE_FOLDER_ID || REASON_EVIDENCE_FOLDER_ID === "여기에_구글드라이브_폴더ID_입력" ) { throw new Error( "StudentReason.gs의 REASON_EVIDENCE_FOLDER_ID를 설정해주세요." ); } const folder = DriveApp.getFolderById( REASON_EVIDENCE_FOLDER_ID ); const safeName = name.replace( /[\\/:*?"<>|]/g, "_" ); const blob = Utilities.newBlob( bytes, type, requestId + "_" + id + "_" + safeName ); const file = folder.createFile(blob); // 관리자만 Drive에서 볼 수 있게 공개 권한은 설정하지 않음. const fileUrl = file.getUrl(); const attachResult = attachReasonEvidence_( id, requestId, fileUrl ); if ( !attachResult || !attachResult.success ) { file.setTrashed(true); return ( attachResult || { success: false, message: "증빙서류 연결에 실패했습니다." } ); } return { success: true, message: "증빙서류가 제출되었습니다.", fileUrl: fileUrl }; }
